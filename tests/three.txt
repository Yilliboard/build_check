from pathlib import Path


def extract_queries_block(text):
    lines = text.splitlines()
    result = []
    inside = False
    base_indent = None

    for line in lines:
        if line.strip().startswith("queries:"):
            inside = True
            base_indent = len(line) - len(line.lstrip())
            result.append(line)
            continue

        if inside:
            indent = len(line) - len(line.lstrip())
            if line.strip() and indent <= base_indent:
                break  # exited queries block
            result.append(line)

    return "\n".join(result)


# Read input as raw text
input_text = Path("config.yaml").read_text()

queries_block = extract_queries_block(input_text)

# Write to Python file
output = f"""# Auto-generated Python config
# Preserves SQL indentation exactly

CONFIG = {{
{queries_block}
}}
"""

Path("config.py").write_text(output)

print("config.py written successfully")
