from pathlib import Path
import re

input_path = Path("config.yaml")
output_path = Path("config.py")

lines = input_path.read_text().splitlines()

out = []
indent_stack = []
in_sql = False
sql_lines = []
current_indent = ""

for line in lines:
    stripped = line.strip()

    # Top-level key (datapre:)
    if stripped.endswith(":") and not indent_stack:
        key = stripped[:-1]
        out.append(f'{key} = {{')
        indent_stack.append("root")
        continue

    # Nested mapping key (queries:)
    if stripped.endswith(":") and not in_sql:
        indent = " " * (len(line) - len(line.lstrip()))
        key = stripped[:-1]
        out.append(f'{indent}"{key}": {{')
        indent_stack.append("map")
        continue

    # Query start: queryX: f'''
    m = re.match(r"(\s*)(\w+):\s*f'''(.*)", line)
    if m:
        current_indent, key, rest = m.groups()
        out.append(f'{current_indent}"{key}": """')
        in_sql = True
        sql_lines = []
        if rest.strip():
            sql_lines.append(rest)
        continue

    # Inside SQL block
    if in_sql:
        if stripped.endswith("'''"):
            content = stripped[:-3]
            if content:
                sql_lines.append(content)
            for s in sql_lines:
                out.append(f"{current_indent}{s.rstrip()}")
            out.append(f'{current_indent}""",')
            in_sql = False
        else:
            sql_lines.append(line.lstrip())
        continue

# Close opened dictionaries
out.append("   }")
out.append("}")

output_path.write_text("\n".join(out))
print("config.py generated successfully")
